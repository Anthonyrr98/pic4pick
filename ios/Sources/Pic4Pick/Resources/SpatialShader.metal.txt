#include <metal_stdlib>
using namespace metal;

struct SpatialVertex {
    float4 position [[position]];
    float2 uv;
};

vertex SpatialVertex spatialVertex(uint index [[vertex_id]]) {
    const float2 positions[] = {
        float2(-1, -1), float2(1, -1), float2(-1, 1),
        float2(-1, 1), float2(1, -1), float2(1, 1)
    };
    float2 p = positions[index];
    SpatialVertex out;
    out.position = float4(p, 0, 1);
    out.uv = float2((p.x + 1) * 0.5, (1 - p.y) * 0.5);
    return out;
}

fragment float4 spatialFragment(SpatialVertex in [[stage_in]],
                               texture2d<float> photo [[texture(0)]],
                               texture2d<float> depth [[texture(1)]],
                               constant float2 &offset [[buffer(0)]]) {
    constexpr sampler linearSampler(coord::normalized, address::clamp_to_edge, filter::linear);
    // Match the website's 1.045 edge allowance and three inverse reprojection steps.
    float2 base = (in.uv - 0.5) / 1.045 + 0.5;
    float2 uv = base;
    for (int i = 0; i < 3; i++) {
        float nearDepth = depth.sample(linearSampler, uv).r;
        uv = base - offset * (nearDepth - 0.35);
    }
    return float4(photo.sample(linearSampler, clamp(uv, float2(0.001), float2(0.999))).rgb, 1);
}
